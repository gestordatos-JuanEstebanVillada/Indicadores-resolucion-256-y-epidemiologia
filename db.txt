CREATE TABLE INTRANET_CALIDAD_VERSIONES (
  ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
  CALIDAD_ID NUMBER NOT NULL,
  INDICADOR VARCHAR2(4000) NOT NULL,
  FORMULA VARCHAR2(4000),
  META_INSTITUCIONAL VARCHAR2(4000),
  OBSERVATORIO VARCHAR2(4000),
  MODULO VARCHAR2(50),
  MES_INICIO NUMBER(2) NOT NULL,
  MES_FIN NUMBER(2) NOT NULL,
  CREATED_AT TIMESTAMP DEFAULT SYSTIMESTAMP,
  UPDATED_AT TIMESTAMP DEFAULT SYSTIMESTAMP,
  CONSTRAINT FK_CALIDAD_VERSION
    FOREIGN KEY (CALIDAD_ID) REFERENCES INTRANET_CALIDAD(ID),
  CONSTRAINT CK_VERSION_MESES
    CHECK (MES_INICIO BETWEEN 1 AND 12 AND MES_FIN BETWEEN MES_INICIO AND 12)
);
